  Lion: ["Antelope", "Zebra", "Boar"],
  Tiger: ["Deer", "Boar", "Monkey", "Cow"],
  Leopard: ["Antelope", "Hare", "Monkey", "Mouse"],

const categories = [
  "Tiny Survivors",
  "Survival Geniuses",
  "Other",
  "Apex Predators",
  "Tough Defenders",
  "Speed Demons",
  "Flight Masters",
  "Underwater Specialists",
  "Stealth & Shadows",
  "Pack Hunters & Social Strategists",
  "Cold-Climate Survivors"
];

const center = { x: 1100, y: 475 };
const radius = 300;

const AnimalMindMap = () => {
  const [hoveredCategory, setHoveredCategory] = useState(null);
  const [hoveredAnimal, setHoveredAnimal] = useState(null);
  const [hoveredAnimalPosition, setHoveredAnimalPosition] = useState(null);
  const [relationshipLines, setRelationshipLines] = useState(null);

  const getAnimalByName = (name) => animalData.find((a) => a.name === name);
  const spacing = 60;

  // Function to get category of hovered animal
  const getHoveredCategory = () => {
    const animal = animalData.find(a => a.name === hoveredAnimal);
    return animal ? animal.category : null;
  };

  // Function to generate relationship lines
  const generateRelationshipLines = (animal, position) => {
    if (!animal || !position) return null;
    const { ax, ay } = position;
    
    return (
      <>
        {/* Lines from predator to its prey */}
        {predatorPrey[animal.name]?.map((prey) => {
          const preyAnimal = getAnimalByName(prey);
          if (!preyAnimal) return null;

          const preyCategoryIndex = categories.indexOf(preyAnimal.category);
          const preyAngle = (2 * Math.PI * preyCategoryIndex) / categories.length;
          const preyCx = center.x + radius * Math.cos(preyAngle);
          const preyCy = center.y + radius * Math.sin(preyAngle);

          // Find the prey's position in its category
          const preyAnimalsInCategory = animalData.filter(a => a.category === preyAnimal.category);
          const preyIndex = preyAnimalsInCategory.findIndex(a => a.name === prey);
          const rowIndex = Math.floor(preyIndex / 6);
          const indexInRow = preyIndex % 6;
          const baseArcRadius = 60 + 20 + 10 + (rowIndex * 50);
          const angleSpacing = Math.PI / 8;
          const animalsInThisRow = Math.min(6, preyAnimalsInCategory.length - rowIndex * 6);
          const startAngle = preyAngle - (angleSpacing * (animalsInThisRow - 1)) / 2;
          const preyAnimalAngle = startAngle + indexInRow * angleSpacing;
          
          const preyAx = preyCx + baseArcRadius * Math.cos(preyAnimalAngle);
          const preyAy = preyCy + baseArcRadius * Math.sin(preyAnimalAngle);

          // Calculate control points for a curved path
          const dx = preyAx - ax;
          const dy = preyAy - ay;
          const distance = Math.sqrt(dx * dx + dy * dy);
          
          // Adjust curve height based on distance
          const curveHeight = distance * 0.3;
          
          // Calculate midpoint
          const midX = (ax + preyAx) / 2;
          const midY = (ay + preyAy) / 2;
          
          // Calculate perpendicular vector for control point
          const perpX = -dy / distance;
          const perpY = dx / distance;
          
          // Control point
          const controlX = midX + perpX * curveHeight;
          const controlY = midY + perpY * curveHeight;

          return (
            <path
              key={`predator-${animal.name}-to-${prey}`}
              d={`M${ax},${ay} Q${controlX},${controlY} ${preyAx},${preyAy}`}
              stroke="#E74C3C"
              strokeOpacity="0.7"
              fill="none"
              strokeWidth="2"
            />
          );
        })}

        {/* Lines from prey to this animal when it's their predator */}
        {Object.entries(predatorPrey).map(([predator, preyList]) => {
          if (!preyList.includes(animal.name)) return null;
          const predatorAnimal = getAnimalByName(predator);
          if (!predatorAnimal) return null;

          const predatorCategoryIndex = categories.indexOf(predatorAnimal.category);
          const predatorAngle = (2 * Math.PI * predatorCategoryIndex) / categories.length;
          const predatorCx = center.x + radius * Math.cos(predatorAngle);
          const predatorCy = center.y + radius * Math.sin(predatorAngle);

          // Find predator's position in its category
          const predatorAnimalsInCategory = animalData.filter(a => a.category === predatorAnimal.category);
          const predatorIndex = predatorAnimalsInCategory.findIndex(a => a.name === predator);
          const rowIndex = Math.floor(predatorIndex / 6);
          const indexInRow = predatorIndex % 6;
          const baseArcRadius = 60 + 20 + 10 + (rowIndex * 50);
          const angleSpacing = Math.PI / 8;
          const animalsInThisRow = Math.min(6, predatorAnimalsInCategory.length - rowIndex * 6);
          const startAngle = predatorAngle - (angleSpacing * (animalsInThisRow - 1)) / 2;
          const predatorAnimalAngle = startAngle + indexInRow * angleSpacing;
          
          const predatorAx = predatorCx + baseArcRadius * Math.cos(predatorAnimalAngle);
          const predatorAy = predatorCy + baseArcRadius * Math.sin(predatorAnimalAngle);

          // Calculate control points for a curved path
          const dx = predatorAx - ax;
          const dy = predatorAy - ay;
          const distance = Math.sqrt(dx * dx + dy * dy);
          
          // Adjust curve height based on distance
          const curveHeight = distance * 0.3;
          
          // Calculate midpoint
          const midX = (ax + predatorAx) / 2;
          const midY = (ay + predatorAy) / 2;
          
          // Calculate perpendicular vector for control point
          const perpX = -dy / distance;
          const perpY = dx / distance;
          
          // Control point
          const controlX = midX + perpX * curveHeight;
          const controlY = midY + perpY * curveHeight;

          return (
            <path
              key={`prey-${animal.name}-to-${predator}`}
              d={`M${ax},${ay} Q${controlX},${controlY} ${predatorAx},${predatorAy}`}
              stroke="#3C97E7"
              strokeOpacity="0.7"
              fill="none"
              strokeWidth="2"
            />
          );
        })}
      </>
    );
  };

  // Effect to update relationship lines when an animal is hovered
  useEffect(() => {
    const hoveredAnimalData = animalData.find(a => a.name === hoveredAnimal);
    setRelationshipLines(generateRelationshipLines(hoveredAnimalData, hoveredAnimalPosition));
  }, [hoveredAnimal, hoveredAnimalPosition]);

  return (
    <svg width="100%" height="1100" style={{ background: "#f0f4f8" }}>
      {categories.map((category, i) => {
        const angle = (2 * Math.PI * i) / categories.length;
        const cx = center.x + radius * Math.cos(angle);
        const cy = center.y + radius * Math.sin(angle);
        const animals = animalData.filter((a) => a.category === category);
        const isHoveredCategory = category === getHoveredCategory();

        return (
          <g
            key={category}
            onMouseEnter={() => setHoveredCategory(category)}
            onMouseLeave={() => setHoveredCategory(null)}
          >
            <line x1={center.x} y1={center.y} x2={cx} y2={cy} stroke="#ccc" />
            <circle 
              cx={cx} 
              cy={cy} 
              r="60" 
              fill={isHoveredCategory ? "#D4ECFF" : "#C9CFD5"} 
            />
            <text x={cx} y={cy} textAnchor="middle" dy=".3em" fontSize="14" fill="#2d3436">
              {category}
            </text>

            {animals.map((animal, j) => {
              const categoryRadius = 60;
              const animalRadius = 20;
              const maxAnimalsPerRow = 6;
              const rowSpacing = 45;
              
              const rowIndex = Math.floor(j / maxAnimalsPerRow);
              const indexInRow = j % maxAnimalsPerRow;
              const animalsInThisRow = Math.min(maxAnimalsPerRow, animals.length - rowIndex * maxAnimalsPerRow);
              
              const baseArcRadius = categoryRadius + animalRadius + 10 + (rowIndex * rowSpacing);
              const angleSpacing = Math.PI / 8;
              const rowArcLength = angleSpacing * (animalsInThisRow - 1);
              const startAngle = angle - rowArcLength / 2;
              const animalAngle = startAngle + indexInRow * angleSpacing;

              const ax = cx + baseArcRadius * Math.cos(animalAngle);
              const ay = cy + baseArcRadius * Math.sin(animalAngle);
              const isHovered = hoveredAnimal === animal.name;

              return (
                <g
                  key={animal.name}
                  onMouseEnter={() => {
                    setHoveredAnimal(animal.name);
                    setHoveredAnimalPosition({ ax, ay });
                  }}
                  onMouseLeave={() => {
                    setHoveredAnimal(null);
                    setHoveredAnimalPosition(null);
                  }}
                >
                  {isHovered ? (
                    <motion.g
                      initial={{ scale: 1 }}
                      animate={{ scale: 1.2 }}
                      style={{ transformOrigin: `${ax}px ${ay}px` }}
                    >
                      <rect 
                        x={ax - 45} 
                        y={ay - 45} 
                        width="90" 
                        height="90" 
                        rx="8"
                        fill="transparent"
                      />
                      <rect 
                        x={ax - 40} 
                        y={ay - 40} 
                        width="80" 
                        height="80" 
                        rx="8"
                        fill="white"
                        stroke="#a1b0c0ff"
                        strokeWidth="1.5"
                      />
                      <image
                        href={animal.image}
                        x={ax - 35}
                        y={ay - 45}
                        width="70"
                        height="70"
                      />
                      <text 
                        x={ax} 
                        y={ay + 32} 
                        textAnchor="middle" 
                        fontSize="14" 
                        fill="#2d3436"
                        fontWeight="bold"
                      >
                        {animal.name}
                      </text>
                    </motion.g>
                  ) : (
                    <motion.g
                      initial={{ scale: 1 }}
                      animate={{ scale: 1 }}
                      style={{ transformOrigin: `${ax}px ${ay}px` }}
                    >
                      <circle
                        cx={ax}
                        cy={ay}
                        r="25"
                        fill="transparent"
                      />
                      <circle
                        cx={ax}
                        cy={ay}
                        r="20"
                        fill="white"
                        stroke="#dfe6e9"
                        strokeWidth="2"
                      />
                      <image
                        href={animal.image}
                        x={ax - 28}
                        y={ay - 28}
                        width="56"
                        height="56"
                        clipPath="circle(18px at 28px 28px)"
                      />
                    </motion.g>
                  )}
                </g>
              );
            })}
          </g>
        );
      })}

      {/* Central Circle */}
      <circle cx={center.x} cy={center.y} r="120" fill="#34495E" />
      <text x={center.x} y={center.y-13} textAnchor="middle" dy=".3em" fontSize="20" fill="white" fontWeight="bold" >
        Animal Survival
      </text>
      <text x={center.x} y={center.y+13} textAnchor="middle" dy=".3em" fontSize="20" fill="white" fontWeight="bold" >
        Categories
      </text>

      {/* Render relationship lines on top */}
      {relationshipLines}
    </svg>
  );
};

export default AnimalMindMap;
